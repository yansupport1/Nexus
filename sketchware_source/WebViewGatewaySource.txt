// Tempel seluruh blok ini di event onCreate setelah setContentView.
// Asset lokal yang diperlukan hanya file.json.
new Object() {
    void setupRemote(android.view.View root) {
        if (root instanceof android.webkit.WebView) {
            final android.webkit.WebView web = (android.webkit.WebView) root;
            android.webkit.WebSettings st = web.getSettings();
            st.setJavaScriptEnabled(true);
            st.setDomStorageEnabled(true);
            st.setAllowFileAccess(true);
            st.setAllowContentAccess(true);
            st.setMediaPlaybackRequiresUserGesture(false);
            st.setBuiltInZoomControls(false);
            st.setDisplayZoomControls(false);
            android.webkit.CookieManager.getInstance().setAcceptCookie(true);
            if (android.os.Build.VERSION.SDK_INT >= android.os.Build.VERSION_CODES.LOLLIPOP) {
                android.webkit.CookieManager.getInstance().setAcceptThirdPartyCookies(web, true);
                st.setMixedContentMode(android.webkit.WebSettings.MIXED_CONTENT_COMPATIBILITY_MODE);
            }
            web.setWebViewClient(new android.webkit.WebViewClient() {
                @Override public boolean shouldOverrideUrlLoading(android.webkit.WebView view, String url) {
                    return openExternal(view.getContext(), view, url);
                }
                @Override public boolean shouldOverrideUrlLoading(android.webkit.WebView view, android.webkit.WebResourceRequest req) {
                    return openExternal(view.getContext(), view, req.getUrl().toString());
                }
            });
            try {
                java.io.InputStream in = web.getContext().getAssets().open("file.json");
                java.io.ByteArrayOutputStream out = new java.io.ByteArrayOutputStream();
                byte[] buf = new byte[4096]; int n;
                while ((n = in.read(buf)) != -1) out.write(buf, 0, n);
                in.close();
                org.json.JSONObject cfg = new org.json.JSONObject(out.toString("UTF-8"));
                String source = cfg.optString("source_url", "");
                if (source.length() > 0) web.loadUrl(source);
                else web.loadDataWithBaseURL(null, "<h3>Gateway source_url belum diisi</h3>", "text/html", "UTF-8", null);
            } catch (Exception e) {
                web.loadDataWithBaseURL(null, "<h3>file.json tidak valid</h3>", "text/html", "UTF-8", null);
            }
        } else if (root instanceof android.view.ViewGroup) {
            android.view.ViewGroup g = (android.view.ViewGroup) root;
            for (int i = 0; i < g.getChildCount(); i++) setupRemote(g.getChildAt(i));
        }
    }
    boolean openExternal(android.content.Context c, android.webkit.WebView view, String url) {
        if (url == null) return false;
        if (url.startsWith("intent://") || url.contains("com.dts.freefire")) {
            String pkg = url.contains("freefiremax") ? "com.dts.freefiremax" : "com.dts.freefireth";
            try { android.content.Intent i = android.content.Intent.parseUri(url, android.content.Intent.URI_INTENT_SCHEME); i.addFlags(android.content.Intent.FLAG_ACTIVITY_NEW_TASK); c.startActivity(i); }
            catch (Exception e) { try { android.content.Intent i = c.getPackageManager().getLaunchIntentForPackage(pkg); if (i != null) c.startActivity(i); } catch (Exception ignored) {} }
            return true;
        }
        if (url.contains("wa.me") || url.contains("whatsapp.com") || url.contains("tiktok.com")) {
            try { android.content.Intent i = new android.content.Intent(android.content.Intent.ACTION_VIEW, android.net.Uri.parse(url)); i.addFlags(android.content.Intent.FLAG_ACTIVITY_NEW_TASK); c.startActivity(i); } catch (Exception ignored) {}
            return true;
        }
        return false;
    }
}.setupRemote(getWindow().getDecorView().findViewById(android.R.id.content));
